/* ---------------- HELPERS ---------------- */

function computeScore(sev, vuln, loc, time) {
  return sev * WEIGHTS.severity + vuln * WEIGHTS.vulnerability + loc * WEIGHTS.locationRisk + time * WEIGHTS.timeSensitivity;
}
function bandFor(score) {
  if (score >= 4.0) return { label: "CRITICAL", color: "#DC2626", bg: "#FEE2E2" };
  if (score >= 3.0) return { label: "HIGH", color: "#C2410C", bg: "#FFEDD5" };
  return { label: "MODERATE", color: "#A16207", bg: "#FEF9C3" };
}
function elapsed(ts, now) {
  const m = Math.max(0, Math.round((now - ts) / 60000));
  if (m < 1) return "just now";
  if (m === 1) return "1 min ago";
  return ${m} mins ago;
}

function ResponseTracker({ status, compact }) {
  const idx = STEP_ORDER.indexOf(status);
  return (
    <div className="flex items-center w-full">
      {STEP_ORDER.map((s, i) => {
        const done = i < idx || (i === idx && status === "RESOLVED");
        const active = i === idx && status !== "RESOLVED";
        const color = status === "RESOLVED" ? "#16A34A" : "#DC2626";
        return (
          <React.Fragment key={s}>
            <div className="flex flex-col items-center" style={{ width: 40 }}>
              <div
                className="rounded-full flex items-center justify-center shrink-0"
                style={{
                  width: compact ? 16 : 20, height: compact ? 16 : 20,
                  background: done || active ? color : "#E2E8F0",
                  color: done || active ? "white" : "#94A3B8",
                }}
              >
                {done ? <CheckCircle2 size={compact ? 11 : 13} /> : <span className="text-[8px] font-bold">{i + 1}</span>}
              </div>
              <p className="text-[7px] mt-0.5 font-semibold text-center leading-none" style={{ color: done || active ? "#0F172A" : "#94A3B8" }}>
                {STEP_LABEL[s]}
              </p>
            </div>
            {i < STEP_ORDER.length - 1 && (
              <div className="flex-1 h-[2px] -mt-3" style={{ background: i < idx ? color : "#E2E8F0" }} />
            )}
          </React.Fragment>
        );
      })}
    </div>
  );
}

/* ---------------- REPORT FLOW (Teacher + Principal) ---------------- */

function ReportView({ me, roster, incidents, connectivity, setConnectivity, pending, submitReport, myLastIncident, isTeacherRole, onSaveProfile }) {
  const [step, setStep] = useState(1);
  const [cat, setCat] = useState(null);
  const [sub, setSub] = useState(null);
  const [pop, setPop] = useState(null);
  const [showProfile, setShowProfile] = useState(false);
  const [showGps, setShowGps] = useState(false);
  const [mismatch, setMismatch] = useState(null); // {label, risk}
  const [justSent, setJustSent] = useState(false);

  const myRoster = roster.find((r) => r.id === me.rosterId);
  const room = myRoster ? myRoster.room : "Bldg A | Flr 1 | Principal's Office";
  const gps = myRoster ? myRoster.gps : "14.312, 121.114";

  function pick(category, subItem) { setCat(category); setSub(subItem); setStep(2); }
  function pickOther() { setCat(OTHER_GENERAL); setSub(OTHER_GENERAL); setStep(2); }
  function pickPop(p) { setPop(p); setStep(3); }

  function reset() { setStep(1); setCat(null); setSub(null); setPop(null); setMismatch(null); }

  function doSubmit() {
    const locationRisk = mismatch ? mismatch.risk : 2;
    const roomLabel = mismatch ? ${mismatch.label} (reassigned from ${room}) : room;
    const timeSensitivity = 3;
    const severity = sub.severity;
    const vulnerability = pop.vuln;
    const score = computeScore(severity, vulnerability, locationRisk, timeSensitivity);
    submitReport({
      category: cat.id === "other" ? "other" : cat.id,
      categoryLabel: cat.id === "other" ? OTHER_GENERAL.label : cat.label,
      subLabel: sub.label,
      severity, vulnerability, locationRisk, timeSensitivity, score,
      room: roomLabel, reporterName: myRoster ? myRoster.name : me.name,
    });
    setJustSent(true);
    reset();
    setTimeout(() => setJustSent(false), 4500);
  }

  return (
    <div className="flex-1 flex flex-col overflow-hidden">
      <div className="px-3 pt-2.5 pb-1.5 shrink-0">
        <div className="bg-white rounded-xl border border-slate-200 px-2.5 py-2 flex items-center gap-2">
          <MapPin size={15} className="text-red-600 shrink-0" />
          <div className="flex-1 min-w-0">
            <p className="text-[11.5px] font-bold text-slate-800 truncate">{room}</p>
            <p className="text-[8.5px] font-mono text-emerald-600">GPS Locked: {gps}</p>
          </div>
          {isTeacherRole && <button onClick={() => setShowProfile(true)} className="p-1"><Settings size={15} className="text-slate-400" /></button>}
        </div>
        <div className="flex items-center gap-1.5 mt-1.5">
          <button
            onClick={() => setShowGps(true)}
            className="text-[8.5px] font-semibold text-orange-600 bg-orange-50 border border-orange-200 rounded-full px-2 py-0.5"
          >⚠ Simulate location mismatch</button>
          <button
            onClick={() => setConnectivity((c) => !c)}
            className={`ml-auto flex items-center gap-1 text-[8.5px] font-semibold rounded-full px-2 py-0.5 border ${connectivity ? "text-emerald-600 bg-emerald-50 border-emerald-200" : "text-slate-500 bg-slate-100 border-slate-200"}`}
          >{connectivity ? <Wifi size={10} /> : <WifiOff size={10} />} {connectivity ? "Online" : "No Connection"}</button>
        </div>
        {mismatch && <p className="text-[8.5px] text-orange-600 mt-1">📍 Reassigned to: {mismatch.label}</p>}
      </div>

      <div className="flex-1 overflow-y-auto px-3 pb-2">
        {justSent && (
          <div className="bg-emerald-50 border border-emerald-200 rounded-lg px-3 py-2 mb-2">
            <p className="text-[11.5px] font-bold text-emerald-700">🔴 Alert broadcast to Principal, Parent & Responder</p>
            <p className="text-[9px] text-emerald-600">No countdown — tracking begins immediately below.</p>
          </div>
        )}

        {!connectivity && pending.length > 0 && (
          <div className="bg-slate-100 border border-slate-300 rounded-lg px-3 py-2 mb-2">
            <p className="text-[11px] font-bold text-slate-700 flex items-center gap-1"><WifiOff size={12} /> Not sent · No connection</p>
            <p className="text-[9.5px] text-slate-600 mt-0.5">Use the school's emergency steps now (alarm, PA system, runner, landline). {pending.length} report(s) queued and will auto-send once reconnected.</p>
          </div>
        )}

        {step === 1 && (
          <div>
            <p className="text-[10px] font-bold text-slate-500 mb-1.5 uppercase tracking-wide">Step 1 · Category</p>
            <div className="grid grid-cols-2 gap-1.5">
              {CATEGORIES.map((c) => {
                const Icon = c.icon;
                return (
                  <div key={c.id} className="bg-white border border-slate-200 rounded-xl p-2">
                    <div className="flex items-center gap-1 mb-1"><Icon size={13} className="text-red-600" /><p className="text-[10.5px] font-bold text-slate-700">{c.label}</p></div>
                    <div className="space-y-1">
                      {c.subs.map((s) => (
                        <button key={s.id} onClick={() => pick(c, s)} className="w-full text-left text-[9.5px] bg-slate-50 hover:bg-red-50 rounded-md px-1.5 py-1 font-medium text-slate-600">
                          {s.label} <span className="font-mono text-slate-400">(Sev {s.severity})</span>
                        </button>
                      ))}
                    </div>
                  </div>
                );
              })}
            </div>
            <button onClick={pickOther} className="w-full bg-red-600 text-white rounded-xl py-2.5 mt-2 text-[12.5px] font-bold">
              Other / General Emergency <span className="font-mono opacity-80">(Sev 4)</span>
            </button>
          </div>
        )}

        {step === 2 && (
          <div>
            <p className="text-[10px] font-bold text-slate-500 mb-1.5 uppercase tracking-wide">Step 2 · Who's affected</p>
            <p className="text-[10.5px] text-slate-500 mb-2">{cat.label} → {sub.label}</p>
            <div className="space-y-1.5">
              {POPULATIONS.map((p) => (
                <button key={p.id} onClick={() => pickPop(p)} className="w-full flex items-center justify-between bg-white border border-slate-200 rounded-xl px-3 py-2.5">
                  <span className="text-[12px] font-semibold text-slate-700">{p.label}</span>
                  <span className="font-mono text-[10px] text-slate-400">Vuln {p.vuln}</span>
                </button>
              ))}
            </div>
            <button onClick={() => setStep(1)} className="text-[10px] text-slate-400 mt-2 font-semibold">← Back</button>
          </div>
        )}

        {step === 3 && (
          <div>
            <p className="text-[10px] font-bold text-slate-500 mb-1.5 uppercase tracking-wide">Step 3 · Confirm & Submit</p>
            <div className="bg-white border border-slate-200 rounded-xl p-3 space-y-1 mb-2">
              <p className="text-[12px] font-bold text-slate-800">{cat.label} — {sub.label}</p>
              <p className="text-[10.5px] text-slate-500">{pop.label}</p>
              <p className="text-[10.5px] text-slate-500">📍 {mismatch ? mismatch.label : room}</p>
            </div>
            <button onClick={doSubmit} className="w-full bg-red-600 text-white rounded-xl py-3 text-[13.5px] font-extrabold tracking-wide">
              🔴 SUBMIT EMERGENCY ALERT
            </button>
            <button onClick={() => setStep(2)} className="text-[10px] text-slate-400 mt-2 font-semibold">← Back</button>
          </div>
        )}

        {myLastIncident && step === 1 && (
          <div className="bg-white border border-slate-200 rounded-xl p-2.5 mt-2.5">
            <p className="text-[9.5px] font-bold text-slate-500 mb-1 uppercase tracking-wide">Your Last Report</p>
            <WaitingLine status={myLastIncident.status} />
            <div className="mt-2"><ResponseTracker status={myLastIncident.status} compact /></div>
          </div>
        )}
      </div>

      {showProfile && myRoster && (
        <ProfileEditModal entry={myRoster} onClose={() => setShowProfile(false)} onSave={(v) => { onSaveProfile(myRoster.id, v); setShowProfile(false); }} />
      )}
      {showGps && (
        <GpsMismatchModal onClose={() => setShowGps(false)} onConfirm={(choice) => { setMismatch(choice); setShowGps(false); }} />
      )}
    </div>
  );
}

/* ---------------- PRINCIPAL DASHBOARD ---------------- */

function IncidentCard({ inc, now, onAck, onResolve, showActions, roleView }) {
  const band = bandFor(inc.score);
  return (
    <div className="bg-white border border-slate-200 rounded-xl p-2.5 mb-2">
      <div className="flex items-start justify-between gap-2">
        <div className="min-w-0">
          <p className="text-[11.5px] font-bold text-slate-800 truncate">{inc.categoryLabel} — {inc.subLabel}</p>
          <p className="text-[9.5px] text-slate-500 truncate">📍 {inc.room}</p>
          <p className="text-[8.5px] text-slate-400">{inc.reporterName} · {elapsed(inc.timestamp, now)}</p>
        </div>
        <ScoreReadout severity={inc.severity} vulnerability={inc.vulnerability} locationRisk={inc.locationRisk} timeSensitivity={inc.timeSensitivity} score={inc.score} />
      </div>
      {inc.confirmedBy > 1 && (
        <p className="text-[8.5px] font-semibold text-orange-600 mt-1">✔ Confirmed by {inc.confirmedBy} reports</p>
      )}
      <div className="mt-2"><ResponseTracker status={inc.status} compact /></div>
      <div className="mt-1.5"><WaitingLine status={inc.status} /></div>
      <p className="text-[8.5px] text-slate-400 mt-1">Recommended: {ESCALATION[inc.category]}</p>
      {showActions && inc.status !== "RESOLVED" && (
        <div className="flex gap-1.5 mt-2">
          {inc.status !== "EN_ROUTE" && (
            <button onClick={() => onAck(inc.id)} className="flex-1 bg-red-600 text-white rounded-lg py-1.5 text-[10.5px] font-bold">🔴 Acknowledge / En Route</button>
          )}
          <button onClick={() => onResolve(inc.id)} className="flex-1 bg-slate-100 text-slate-600 rounded-lg py-1.5 text-[10.5px] font-bold">Mark Resolved</button>
        </div>
      )}
    </div>

/* ---------------- APP ---------------- */

export default function App() {
  const [me, setMe] = useState(null);
  const [roster, setRoster] = useState(INITIAL_ROSTER);
  const [incidents, setIncidents] = useState(() => ([{
    id: nextId(), category: "flood", categoryLabel: "Flooding/Natural", subLabel: "Flash Flood",
    severity: 4, vulnerability: 4, locationRisk: 2, timeSensitivity: 3,
    score: computeScore(4, 4, 2, 3), room: "Bldg B | Flr 1 | Rm 105", reporterName: "Ms. Bautista",
    timestamp: Date.now() - 6 * 60000, status: "EN_ROUTE", confirmedBy: 2, linkedSection: "Grade 5 - Sec A",
  }]));
  const [connectivity, setConnectivity] = useState(true);
  const [pending, setPending] = useState([]);
  const [now, setNow] = useState(Date.now());

  useEffect(() => { const t = setInterval(() => setNow(Date.now()), 20000); return () => clearInterval(t); }, []);

  useEffect(() => {
    if (connectivity && pending.length > 0) {
      setIncidents((prev) => [...prev, ...pending.map((p) => ({ ...p, status: "SENT", timestamp: Date.now() }))]);
      setPending([]);
    }
  }, [connectivity]); // eslint-disable-line

  useEffect(() => {
    const t = setTimeout(() => {
      setIncidents((prev) => prev.map((i) => (i.status === "SENT" ? { ...i, status: "SEEN" } : i)));
    }, 2200);
    return () => clearTimeout(t);
  }, [incidents.length]);

  function findLinkedSection(room) {
    const r = roster.find((t) => room.startsWith(t.room));
    return r ? r.section : roster[0]?.section;
  }

  function submitReport(data) {
    const windowMs = 10 * 60000;
    const dup = incidents.find((i) => i.status !== "RESOLVED" && i.category === data.category && i.room === data.room && (Date.now() - i.timestamp) < windowMs);
    if (dup) {
      setIncidents((prev) => prev.map((i) => (i.id === dup.id ? { ...i, confirmedBy: (i.confirmedBy || 1) + 1 } : i)));
      return;
    }
    const record = { id: nextId(), ...data, timestamp: Date.now(), status: "SENT", confirmedBy: 1, linkedSection: findLinkedSection(data.room) };
    if (!connectivity) { setPending((p) => [...p, record]); return; }
    setIncidents((prev) => [...prev, record]);
  }

  function ack(id) { setIncidents((prev) => prev.map((i) => (i.id === id ? { ...i, status: "EN_ROUTE" } : i))); }
  function resolve(id) { setIncidents((prev) => prev.map((i) => (i.id === id ? { ...i, status: "RESOLVED" } : i))); }
  function saveProfile(id, vals) { setRoster((prev) => prev.map((t) => (t.id === id ? { ...t, ...vals } : t))); }
  function addTeacher(vals) { setRoster((prev) => [...prev, { id: t${Date.now()}, ...vals }]); }
  function editTeacher(id, vals) { setRoster((prev) => prev.map((t) => (t.id === id ? { ...t, ...vals } : t))); }
  function deleteTeacher(id) { setRoster((prev) => prev.filter((t) => t.id !== id)); }

  function logout() { setMe(null); }

  const myLastIncident = me && me.rosterId
    ? incidents.filter((i) => i.reporterName === roster.find((r) => r.id === me.rosterId)?.name).sort((a, b) => b.timestamp - a.timestamp)[0]
    : (me && me.role === "principal" ? incidents.slice().sort((a, b) => b.timestamp - a.timestamp)[0] : null);

  return (
    <div className="w-full h-full flex items-center justify-center bg-slate-950 font-sans" style={{ minHeight: "100dvh" }}>
      <div className="w-full max-w-[400px] h-[100dvh] max-h-[860px] bg-slate-50 flex flex-col shadow-2xl overflow-hidden relative" style={{ color: "#0F172A" }}>
        {!me && <LoginScreen onLogin={setMe} />}

        {me && me.role === "teacher" && (
          <>
            <Header title="AGAP · Teacher" subtitle={me.subtitle} onLogout={logout} />
            <ReportView me={me} roster={roster} incidents={incidents} connectivity={connectivity} setConnectivity={setConnectivity}
              pending={pending} submitReport={submitReport} myLastIncident={myLastIncident} isTeacherRole onSaveProfile={saveProfile} />
            <Footer />
          </>
        )}

        {me && me.role === "principal" && (
          <>
            <Header title="AGAP · Principal" subtitle={SCHOOL.name} onLogout={logout} />
            <PrincipalDashboard me={me} roster={roster} incidents={incidents} connectivity={connectivity} setConnectivity={setConnectivity}
              pending={pending} submitReport={submitReport} myLastIncident={myLastIncident} ack={ack} resolve={resolve} saveProfile={saveProfile}
              addTeacher={addTeacher} editTeacher={editTeacher} deleteTeacher={deleteTeacher} now={now} />
            <Footer />
          </>
        )}

        {me && me.role === "parent" && (
          <>
            <Header title="AGAP · Parent/PTA" subtitle={me.subtitle} onLogout={logout} />
            <ParentView me={me} incidents={incidents} now={now} />
            <Footer />
          </>
        )}

        {me && me.role === "responder" && (
          <>
            <Header title="AGAP · Responder" subtitle="Barangay Sinalhan + CDRRMO" onLogout={logout} />
            <ResponderView incidents={incidents} ack={ack} resolve={resolve} now={now} />
            <Footer />
          </>
        )}
      </div>
    </div>
  );
}
